% Word-like paragraph behavior (tighter)
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.18\baselineskip}
\setstretch{1.03}
\raggedbottom

% Lists (Word-like indentation and tight spacing)
\setlist[itemize]{leftmargin=1.25em, itemsep=0.15em, topsep=0.15em, parsep=0pt, partopsep=0pt}
\setlist[enumerate]{leftmargin=1.25em, itemsep=0.15em, topsep=0.15em, parsep=0pt, partopsep=0pt}

% -----------------------------
% Gray boxed heading bar (Word-like)
% -----------------------------
\definecolor{uiclightgray}{RGB}{200,200,200}

% Title (Pandoc `# ...`) should be big and centered (no gray bar)
\titleformat{\section}{\Large\bfseries\sffamily\centering}{}{-0.25em}{#1}
\titlespacing*{\section}{0pt}{0.2\baselineskip}{0.7\baselineskip}

% Subsections: gray heading bar (Pandoc emits \subsection for `##` headings)
\titleformat{\subsection}
  {\Large\bfseries\sffamily}
  {}
  {0em}
  {%
    \begin{tikzpicture}[baseline=(title.base)]
      \node[
        inner sep=8pt,
        fill=uiclightgray,
        draw=black,
        line width=0.8pt,
        text width=\dimexpr\textwidth-16pt\relax,
        align=left,
        anchor=west
      ] (title) {#1};
    \end{tikzpicture}%
  }
  []

\titlespacing*{\subsection}{0pt}{0.9\baselineskip}{0.6\baselineskip}

% Subsubsections: compact and bold (no bar)
\titleformat{\subsubsection}{\bfseries}{\thesubsubsection}{0.5em}{#1}
\titlespacing*{\subsubsection}{0pt}{0.8\baselineskip}{0.35\baselineskip}

% -----------------------------
% Table helpers
% -----------------------------
